import numpy as np
import matplotlib.pyplot as plt

def pure_cnvs_monte_carlo(M_fragments=100, lambda_factor=1.0, rho_c=0.05, trials=5000):
    """
    M_fragments: Number of critical tasks/fragments for the Global Veto.
    lambda_factor: Fragmentation multiplier (redundancy).
    rho_c: Dependent collusion parameter (correlation density).
    """
    N_total = int(M_fragments * lambda_factor)
    q_values = np.linspace(0.01, 0.99, 50) # Percentage of nodes corrupted by the attacker (from 1% to 99%)
    p_success = []

    for q in q_values:
        wins = 0
        for _ in range(trials):
            # 1. Primary Attack: The attacker randomly corrupts nodes (q)
            # binomial: out of N_total nodes, each has probability q of being corrupted
            primary_compromised = np.random.binomial(N_total, q)
            
            # 2. Secondary Attack (Dependent Collusion / Inference)
            remaining_nodes = N_total - primary_compromised
            
            # The probability of inferring a remaining node depends on how many nodes you already have and on rho_c
            # P_inf = 1 - (1 - rho_c)^(possessed nodes)
            p_inference = 1.0 - (1.0 - rho_c)**primary_compromised
            
            # The attacker infers a certain number of remaining nodes
            secondary_compromised = np.random.binomial(remaining_nodes, p_inference)
            
            total_compromised = primary_compromised + secondary_compromised
            
            # 3. Winning Condition (Global Veto)
            # The attacker wins ONLY IF they hold at least all the necessary M_fragments
            if total_compromised >= M_fragments:
                wins += 1
                
        p_success.append(wins / trials)
        
    return q_values, p_success

# --- TEST EXECUTION ---
plt.figure(figsize=(10, 6))

# We fix rho_c to a constant value and vary lambda
for lam in [1.0, 1.2, 1.5, 2.0, 2.5, 3.0, 3.5, 4.0, 4.5, 5.0, 6.0, 7.0, 10.0, 14.0]:
    q_vals, p_vals = pure_cnvs_monte_carlo(M_fragments=50, lambda_factor=lam, rho_c=0.01)
    plt.plot(q_vals, p_vals, label=f'lambda = {lam}')

plt.axvline(x=0.33, color='red', linestyle='--', label='Byzantine Limit (1/3)')
plt.title("CNVS Parametric Test: Probability of False Positive")
plt.xlabel("Fraction of Initially Corrupted Nodes (q)")
plt.ylabel("Attacker Win Probability (False Positive)")
plt.legend()
plt.grid(True)
plt.show()
